cmake_minimum_required(VERSION 3.16)
project(omx_turtlebot3_control_cpp)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(urdf REQUIRED)
find_package(OpenCV 4 REQUIRED COMPONENTS core dnn)
find_package(OpenSSL REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
# v6 actor (OpenCV DNN on the original ONNX), model checks and residual step.
add_library(omx_v6_control SHARED src/control.cpp)
target_compile_features(omx_v6_control PUBLIC cxx_std_17)
target_compile_options(omx_v6_control PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(omx_v6_control PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_include_directories(omx_v6_control SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(omx_v6_control urdf)
target_link_libraries(omx_v6_control ${OpenCV_LIBS} OpenSSL::Crypto yaml-cpp)
# Startup check: ONNX output against the frozen parity vectors.
add_executable(onnx_probe_cpp src/probe.cpp)
target_link_libraries(onnx_probe_cpp omx_v6_control)
# Adds the v6 residual to mp_control's arm trajectories.
add_executable(ros_policy_relay_cpp src/ros_policy_relay.cpp)
target_link_libraries(ros_policy_relay_cpp omx_v6_control)
ament_target_dependencies(ros_policy_relay_cpp rclcpp sensor_msgs std_msgs trajectory_msgs)
install(TARGETS omx_v6_control EXPORT omx_v6_control LIBRARY DESTINATION lib)
install(TARGETS onnx_probe_cpp ros_policy_relay_cpp DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(omx_v6_control HAS_LIBRARY_TARGET)
ament_export_dependencies(urdf OpenCV OpenSSL yaml-cpp)
ament_package()
